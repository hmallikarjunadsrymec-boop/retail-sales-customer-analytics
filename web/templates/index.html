Production DAX Measures Library (Preview)

// 1. Total Gross Revenue
Total Revenue = SUM(FactSales[Revenue])

// 2. Average Order Value
Average Order Value = DIVIDE([Total Revenue], DISTINCTCOUNT(FactSales[InvoiceNo]), 0)

// 3. Month-over-Month Revenue Growth (%)
MoM Revenue Growth Pct = 
VAR CurrentRevenue = [Total Revenue]
VAR PriorRevenue = CALCULATE([Total Revenue], PREVIOUSMONTH(DimDate[FullDate]))
RETURN DIVIDE(CurrentRevenue - PriorRevenue, PriorRevenue, 0)

// 4. Repeat Customer Retention Rate (%)
Customer Retention Rate = 
DIVIDE(
    COUNTROWS(FILTER(VALUES(DimCustomer[CustomerID]), CALCULATE(DISTINCTCOUNT(FactSales[InvoiceNo])) > 1)),
    CALCULATE(DISTINCTCOUNT(FactSales[CustomerID]), FactSales[CustomerID] <> "GUEST"),
    0
)
